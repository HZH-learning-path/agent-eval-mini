### 2.3 写 runner.py —— 评测框架核心

这是最重要的文件，大约 150 行，分三部分：调模型、执行任务、自动评分。

```python
# runner.py
import json
import os
import zipfile
import shutil
from openai import OpenAI
from config import API_KEY, API_BASE, MODEL_NAME, EXPERIMENTS

client = OpenAI(api_key=API_KEY, base_url=API_BASE)


def setup_task(task, work_dir):
    """根据任务定义创建初始文件"""
    os.makedirs(work_dir, exist_ok=True)
    files = task.get("setup", {}).get("files", {})
    for filepath, content in files.items():
        full_path = os.path.join(work_dir, filepath)
        os.makedirs(os.path.dirname(full_path), exist_ok=True)

        if content.startswith("zip:"):
            # 处理zip文件：格式 zip:文件名=内容
            zip_info = content[4:]
            filename, file_content = zip_info.split("=", 1)
            with zipfile.ZipFile(full_path, "w") as zf:
                zf.writestr(filename, file_content)
        else:
            with open(full_path, "w", encoding="utf-8") as f:
                f.write(content)


def call_model(prompt, max_tokens):
    """调用大模型API"""
    response = client.chat.completions.create(
        model=MODEL_NAME,
        messages=[{"role": "user", "content": prompt}],
        max_tokens=max_tokens,
        temperature=0,
    )
    return response.choices[0].message.content


def check_result(task, model_output, work_dir):
    """自动评分：检查模型输出是否正确"""
    check = task["check"]

    if check["type"] == "contains":
        # 检查模型输出中是否包含正确答案
        passed = check["answer"].lower() in model_output.lower()
        return {
            "passed": passed,
            "detail": f"期望包含 '{check['answer']}'，实际输出: {model_output[:200]}"
        }

    elif check["type"] == "file_exists":
        # 检查文件是否被创建且内容正确
        file_path = os.path.join(work_dir, check["path"])
        if not os.path.exists(file_path):
            return {"passed": False, "detail": f"文件不存在: {check['path']}"}
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read().strip()
        passed = check["answer"] in content
        return {
            "passed": passed,
            "detail": f"文件内容: '{content}'，期望: '{check['answer']}'"
        }

    return {"passed": False, "detail": "未知的检查类型"}


def run_experiment(exp_name, exp_config, tasks):
    """跑一组实验"""
    print(f"\n{'='*50}")
    print(f"开始实验: {exp_config['label']}")
    print(f"说明: {exp_config['description']}")
    print(f"{'='*50}")

    work_dir = exp_config["work_dir"]
    max_tokens = exp_config["max_tokens"]

    # 清理工作目录
    if os.path.exists(work_dir):
        shutil.rmtree(work_dir)
    if work_dir == "./wrong_workspace":
        # C组：故意不创建目录，让文件操作失败
        pass
    else:
        os.makedirs(work_dir, exist_ok=True)

    results = []
    for task in tasks:
        print(f"\n--- {task['id']}: {task['title']} ---")

        # 1. 准备初始文件
        if work_dir != "./wrong_workspace":
            setup_task(task, work_dir)

        # 2. 调用模型
        try:
            model_output = call_model(task["prompt"], max_tokens)
        except Exception as e:
            model_output = f"[API调用失败] {str(e)}"

        print(f"模型输出: {model_output[:100]}...")

        # 3. 对于文件操作题，模型会输出代码而不是直接执行
        #    这里我们简化处理：检查模型输出内容
        #    （真实Harness中模型会通过工具调用来操作文件）
        check = check_result(task, model_output, work_dir)

        result = {
            "task_id": task["id"],
            "task_title": task["title"],
            "experiment": exp_name,
            "max_tokens": max_tokens,
            "work_dir": work_dir,
            "model_output": model_output,
            "passed": check["passed"],
            "detail": check["detail"]
        }
        results.append(result)
        print(f"结果: {'✅ 通过' if check['passed'] else '❌ 失败'} - {check['detail']}")

    return results


def main():
    # 加载所有任务
    tasks = []
    task_dir = "tasks"
    for i in range(1, 5):
        with open(os.path.join(task_dir, f"task{i}.json"), "r", encoding="utf-8") as f:
            tasks.append(json.load(f))

    # 跑三组实验
    all_results = {}
    for exp_name, exp_config in EXPERIMENTS.items():
        all_results[exp_name] = run_experiment(exp_name, exp_config, tasks)

    # 保存结果
    os.makedirs("results", exist_ok=True)
    timestamp = "experiment_results"
    result_file = os.path.join("results", f"{timestamp}.json")
    with open(result_file, "w", encoding="utf-8") as f:
        json.dump(all_results, f, ensure_ascii=False, indent=2)

    # 打印汇总
    print(f"\n{'='*50}")
    print("实验汇总")
    print(f"{'='*50}")
    for exp_name, results in all_results.items():
        passed = sum(1 for r in results if r["passed"])
        total = len(results)
        label = EXPERIMENTS[exp_name]["label"]
        print(f"{label}: {passed}/{total} 通过 ({passed/total*100:.0f}%)")

    print(f"\n详细结果已保存到: {result_file}")


if __name__ == "__main__":
    main()
```

## 三、Day 2：设计并跑对照实验

### 3.1 确认三组实验的逻辑

| 组 | 改了什么 | 模型能力变了吗 | 预期分数变化 |
|:---|:---|:---|:---|
| A正常 | 无（基线） | - | 100% |
| B掐token | max_tokens=40 | 没有，模型还是那个模型 | 大幅下降 |
| C错路径 | work_dir指向不存在目录 | 没有，模型还是那个模型 | 大幅下降 |

**核心论点**：B和C分数低，但不是模型变笨了，是执行层配置的锅。

### 3.2 跑实验并记录

运行 `python runner.py` 后，检查 `results/experiment_results.json`，确认每道题的：
- 模型输出内容
- 通过/失败状态
- 失败原因

### 3.3 如果结果不理想怎么调

**问题：B组通过率没下降？**
→ 检查题目是否太简单（不依赖长输出）。可以加一道"请输出10个用户的名字"这种需要长输出的题。

**问题：C组通过率没下降？**
→ 检查 work_dir 路径是否真的不存在。task1（读JSON）依赖文件，应该会失败；task4（纯文本推理）不依赖文件，应该通过——这恰好说明"不是所有题都受执行层影响"，本身就是好的发现。

**问题：API报错？**
→ 检查 API Key 是否设置、账户余额是否充足、网络是否正常。

## 四、Day 3：分析 + 写报告

### 4.1 运行分析脚本

写一个 `analyze.py`，自动生成报告的核心数据：

```python
# analyze.py
import json
import os

def main():
    with open("results/experiment_results.json", "r", encoding="utf-8") as f:
        data = json.load(f)

    print("# Agent 评测对照实验报告\n")
    print("## 一、实验设计\n")
    print("| 组别 | 配置 | 改了什么 |")
    print("|:---|:---|:---|")
    print("| A组（正常） | max_tokens=2048, 正确路径 | 基线 |")
    print("| B组（掐token） | max_tokens=40, 正确路径 | 仅改token限制 |")
    print("| C组（错路径） | max_tokens=2048, 错误路径 | 仅改文件路径 |")
    print()

    print("## 二、实验结果\n")
    print("| 题目 | A组（正常） | B组（掐token） | C组（错路径） |")
    print("|:---|:---|:---|:---|")

    task_ids = [r["task_id"] for r in data["A_normal"]]
    for tid in task_ids:
        row = f"| {tid} |"
        for exp_key in ["A_normal", "B_token_limit", "C_wrong_path"]:
            r = next(x for x in data[exp_key] if x["task_id"] == tid)
            row += f" {'✅' if r['passed'] else '❌'} |"
        print(row)

    print("\n| 组别 | 通过率 |")
    print("|:---|:---|")
    for exp_key, label in [("A_normal", "A组"), ("B_token_limit", "B组"), ("C_wrong_path", "C组")]:
        passed = sum(1 for r in data[exp_key] if r["passed"])
        total = len(data[exp_key])
        print(f"| {label} | {passed}/{total} ({passed/total*100:.0f}%) |")

    print("\n## 三、失败案例分析\n")
    for exp_key, label in [("B_token_limit", "B组"), ("C_wrong_path", "C组")]:
        print(f"### {label}失败案例\n")
        for r in data[exp_key]:
            if not r["passed"]:
                print(f"**{r['task_id']}**: {r['detail']}")
                print(f"  - 模型输出: {r['model_output'][:150]}")
                print()
        print()

    print("## 四、归因结论\n")
    print("1. **B组失败归因：执行层token限制**")
    print("   - 模型知道答案，但输出被截断")
    print("   - 这是执行层（Harness）的问题，不是模型能力问题\n")
    print("2. **C组失败归因：执行层文件路径错误**")
    print("   - 模型给出了正确的操作指令，但文件路径不存在导致执行失败")
    print("   - 同样是执行层问题\n")
    print("3. **核心发现**")
    print("   - 同一个模型，仅改变执行层配置，通过率可下降50%+")
    print("   - 如果只看通过率，会误判为'模型能力差'")
    print("   - Agent评测必须拆分模型层和执行层，否则分数没有意义\n")

    print("## 五、实验局限\n")
    print("- 任务数量较少（4题），统计意义有限")
    print("- 文件操作题中模型输出的是代码而非直接执行，与真实Agent有差异")
    print("- 仅测试了一个模型，未做跨模型验证")
    print("- 未来方向：接入真实Agent框架（如DeepSeek Harness），验证归因方法\n")


if __name__ == "__main__":
    main()
```
